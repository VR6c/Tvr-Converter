function fftInPlace(re: Float32Array, im: Float32Array) {
  const n = re.length;
  // bit reversal
  for (let i = 1, j = 0; i < n; i++) {
    let bit = n >> 1;
    for (; j & bit; bit >>= 1) j ^= bit;
    j ^= bit;
    if (i < j) { [re[i], re[j]] = [re[j], re[i]]; [im[i], im[j]] = [im[j], im[i]]; }
  }
  for (let len = 2; len <= n; len <<= 1) {
    const ang = -2 * Math.PI / len;
    const wRe = Math.cos(ang), wIm = Math.sin(ang);
    for (let i = 0; i < n; i += len) {
      let curRe = 1, curIm = 0;
      for (let k = 0; k < len / 2; k++) {
        const uRe = re[i+k], uIm = im[i+k];
        const vRe = re[i+k+len/2]*curRe - im[i+k+len/2]*curIm;
        const vIm = re[i+k+len/2]*curIm + im[i+k+len/2]*curRe;
        re[i+k] = uRe + vRe; im[i+k] = uIm + vIm;
        re[i+k+len/2] = uRe - vRe; im[i+k+len/2] = uIm - vIm;
        const nRe = curRe*wRe - curIm*wIm;
        curIm = curRe*wIm + curIm*wRe;
        curRe = nRe;
      }
    }
  }
}

function yin(frame: Float32Array, sr: number, fmin: number, fmax: number) {
  const tauMin = Math.max(2, Math.floor(sr / fmax));
  const tauMax = Math.min(frame.length - 1, Math.ceil(sr / fmin));
  const W = frame.length - tauMax;
  if (W <= 0) return { f0: 0, conf: 0 };
  const d = new Float32Array(tauMax + 1);
  for (let tau = tauMin; tau <= tauMax; tau++) {
    let sum = 0;
    for (let i = 0; i < W; i++) { const diff = frame[i] - frame[i + tau]; sum += diff * diff; }
    d[tau] = sum;
  }
  // CMND
  let running = 0;
  const cmnd = new Float32Array(tauMax + 1);
  for (let tau = 1; tau <= tauMax; tau++) {
    running += d[tau];
    cmnd[tau] = running === 0 ? 1 : (d[tau] * tau) / running;
  }
  const threshold = 0.15;
  let best = -1;
  for (let tau = tauMin; tau <= tauMax; tau++) {
    if (cmnd[tau] < threshold) {
      while (tau + 1 <= tauMax && cmnd[tau + 1] < cmnd[tau]) tau++;
      best = tau; break;
    }
  }
  if (best === -1) {
    // global min
    let m = Infinity;
    for (let tau = tauMin; tau <= tauMax; tau++) if (cmnd[tau] < m) { m = cmnd[tau]; best = tau; }
    if (m > 0.35) return { f0: 0, conf: 0 };
  }
  // parabolic
  let better = best;
  if (best > tauMin && best < tauMax) {
    const s0 = cmnd[best-1], s1 = cmnd[best], s2 = cmnd[best+1];
    const denom = 2 * (2*s1 - s2 - s0);
    if (denom !== 0) better = best + (s2 - s0) / denom;
  }
  const f0 = sr / better;
  if (f0 < fmin || f0 > fmax) return { f0: 0, conf: 0 };
  return { f0, conf: Math.max(0, Math.min(1, 1 - cmnd[best])) };
}

/**

  • Downmix to mono and resample to exactly 22,050 Hz using

  • OfflineAudioContext(1, targetFrames, 22050).
    */
    async function resampleMono(
    buffer: AudioBuffer,
    targetRate = TARGET_SAMPLE_RATE,
    ): Promise<Float32Array> {
    const duration = buffer.duration;
    const targetFrames = Math.max(1, Math.ceil(duration * targetRate));

const offline = new OfflineAudioContext(1, targetFrames, targetRate);
const source = offline.createBufferSource();
source.buffer = buffer;

// Explicit downmix graph: multi-channel → mono.
const merger = offline.createGain();
merger.gain.value = 1;
source.connect(merger);
merger.connect(offline.destination);
source.start(0);

const rendered = await offline.startRendering();
return rendered.getChannelData(0).slice();
}

/** Build a min/max peak envelope for fast waveform rendering. */
export function buildPeaks(samples: Float32Array, buckets = 2000): Float32Array {
const out = new Float32Array(buckets * 2);
const step = Math.max(1, Math.floor(samples.length / buckets));
for (let b = 0; b < buckets; b++) {
const start = b * step;
let min = 0;
let max = 0;
for (let i = 0; i < step; i++) {
const idx = start + i;
if (idx >= samples.length) break;
const v = samples[idx];
if (v < min) min = v;
if (v > max) max = v;
}
out[b * 2] = min;
out[b * 2 + 1] = max;
}
return out;
}

const samples = await resampleMono(buffer);
return {
samples,
sampleRate: TARGET_SAMPLE_RATE,
duration: samples.length / TARGET_SAMPLE_RATE,
peaks: buildPeaks(samples),
};
}

function correlate(a: number[], b: number[]): number {
const n = a.length;
const ma = a.reduce((s, v) => s + v, 0) / n;
const mb = b.reduce((s, v) => s + v, 0) / n;
let num = 0;
let da = 0;
let db = 0;
for (let i = 0; i < n; i++) {
const x = a[i] - ma;
const y = b[i] - mb;
num += x * y;
da += x * x;
db += y * y;
}
const den = Math.sqrt(da * db);
return den === 0 ? 0 : num / den;
}

/* ------------------------------------------------------------------ */
/* FFT (iterative radix-2) */
/* ------------------------------------------------------------------ */

function fftInPlace(re: Float32Array, im: Float32Array): void {
const n = re.length;

for (let i = 1, j = 0; i < n; i++) {
let bit = n >> 1;
for (; j & bit; bit >>= 1) j ^= bit;
j ^= bit;
if (i < j) {
let t = re[i]; re[i] = re[j]; re[j] = t;
t = im[i]; im[i] = im[j]; im[j] = t;
}
}

for (let len = 2; len <= n; len <<= 1) {
const ang = (-2 * Math.PI) / len;
const wRe = Math.cos(ang);
const wIm = Math.sin(ang);
const half = len >> 1;

for (let i = 0; i < n; i += len) {
let curRe = 1;
let curIm = 0;
for (let k = 0; k < half; k++) {
const aRe = re[i + k];
const aIm = im[i + k];
const bRe = re[i + k + half] * curRe - im[i + k + half] * curIm;
const bIm = re[i + k + half] * curIm + im[i + k + half] * curRe;

re[i + k] = aRe + bRe;
im[i + k] = aIm + bIm;
re[i + k + half] = aRe - bRe;
im[i + k + half] = aIm - bIm;

const nRe = curRe * wRe - curIm * wIm;
curIm = curRe * wIm + curIm * wRe;
curRe = nRe;
}
}
}
}

/* ------------------------------------------------------------------ */
/* YIN monophonic pitch detection */
/* ------------------------------------------------------------------ */

interface YinResult {
f0: number;
confidence: number;
rms: number;
}

function yinFrame(
frame: Float32Array,
sr: number,
fmin: number,
fmax: number,
): YinResult {
const n = frame.length;
let rms = 0;
for (let i = 0; i < n; i++) rms += frame[i] * frame[i];
rms = Math.sqrt(rms / n);

const tauMin = Math.max(2, Math.floor(sr / fmax));
const tauMax = Math.min(n - YIN_W - 1, Math.ceil(sr / fmin));
if (tauMax <= tauMin) return { f0: 0, confidence: 0, rms };

const d = new Float32Array(tauMax + 1);
for (let tau = tauMin; tau <= tauMax; tau++) {
let sum = 0;
for (let i = 0; i < YIN_W; i++) {
const diff = frame[i] - frame[i + tau];
sum += diff * diff;
}
d[tau] = sum;
}

const cmnd = new Float32Array(tauMax + 1);
let running = 0;
for (let tau = 1; tau <= tauMax; tau++) {
running += d[tau];
cmnd[tau] = running === 0 ? 1 : (d[tau] * tau) / running;
}

const THRESH = 0.15;
let best = -1;
for (let tau = tauMin; tau <= tauMax; tau++) {
if (cmnd[tau] < THRESH) {
while (tau + 1 <= tauMax && cmnd[tau + 1] < cmnd[tau]) tau++;
best = tau;
break;
}
}

if (best === -1) {
let minVal = Infinity;
for (let tau = tauMin; tau <= tauMax; tau++) {
if (cmnd[tau] < minVal) {
minVal = cmnd[tau];
best = tau;
}
}
if (minVal > 0.35) return { f0: 0, confidence: 0, rms };
}

let refined = best;
if (best > tauMin && best < tauMax) {
const s0 = cmnd[best - 1];
const s1 = cmnd[best];
const s2 = cmnd[best + 1];
const denom = 2 * (2 * s1 - s2 - s0);
if (denom !== 0) refined = best + (s2 - s0) / denom;
}

const f0 = sr / refined;
if (!Number.isFinite(f0) || f0 < fmin || f0 > fmax) {
return { f0: 0, confidence: 0, rms };
}

return {
f0,
confidence: Math.max(0, Math.min(1, 1 - cmnd[best])),
rms,
};
}

const hzToMidi = (hz: number) => 69 + 12 * Math.log2(hz / 440);

/* ------------------------------------------------------------------ */
/* Spectral polyphonic salience */
/* ------------------------------------------------------------------ */

function buildPolySalience(
samples: Float32Array,
sampleRate: number,
): { salience: Float32Array; numFrames: number; hopSec: number } {
const hop = POLY_HOP;
const n = POLY_FFT;
const numFrames = Math.max(1, Math.floor((samples.length - n) / hop) + 1);
const salience = new Float32Array(128 * numFrames);
const hopSec = hop / sampleRate;

const window = new Float32Array(n);
for (let i = 0; i < n; i++) {
window[i] = 0.5 * (1 - Math.cos((2 * Math.PI * i) / (n - 1)));
}

const re = new Float32Array(n);
const im = new Float32Array(n);
const mag = new Float32Array(n / 2);

for (let f = 0; f < numFrames; f++) {
const offset = f * hop;
for (let i = 0; i < n; i++) {
re[i] = samples[offset + i] * window[i];
im[i] = 0;
}
fftInPlace(re, im);

let maxMag = 0;
for (let i = 0; i < n / 2; i++) {
mag[i] = Math.hypot(re[i], im[i]);
if (mag[i] > maxMag) maxMag = mag[i];
}
if (maxMag < 1e-6) continue;

const floor = maxMag * 0.008; // ~ -42 dB below the loudest partial

for (let i = 2; i < n / 2 - 1; i++) {
const m = mag[i];
if (m < floor) continue;
if (m <= mag[i - 1] || m < mag[i + 1]) continue; // local maximum only

// Parabolic interpolation for sub-bin accuracy.
const a = mag[i - 1];
const b = m;
const c = mag[i + 1];
const denom = a - 2 * b + c;
const delta = denom !== 0 ? (0.5 * (a - c)) / denom : 0;
const bin = i + delta;
const freq = (bin * sampleRate) / n;
if (freq < 55 || freq > 4200) continue;

const midi = hzToMidi(freq);
const pitch = Math.round(midi);
if (pitch < 21 || pitch > 108) continue;

// Confidence weighted by magnitude and detune distance.
const detune = Math.abs(midi - pitch);
const w = (b / maxMag) * Math.max(0, 1 - detune * 1.6);
const idx = pitch * numFrames + f;
if (w > salience[idx]) salience[idx] = w;
}

if (f % 40 === 0) progress((f / numFrames) * 0.9);
}

return { salience, numFrames, hopSec };
}

function salienceToNotes(
salience: Float32Array,
numFrames: number,
hopSec: number,
onsetThreshold: number,
): TranscribedNote[] {

const out: TranscribedNote[] = [];

for (let p = 21; p <= 108; p++) {
let f = 0;
while (f < numFrames) {
const v = salience[p * numFrames + f];
if (v < onsetThreshold) {
f++;
continue;
}

let j = f;
let peak = v;
let silent = 0;

while (j < numFrames) {
const s = salience[p * numFrames + j];
if (s >= release) {
silent = 0;
if (s > peak) peak = s;
} else {
silent++;
if (silent > 3) break;
}
j++;
}

const endFrame = j - silent;
const duration = (endFrame - f) * hopSec;

if (duration >= MIN_NOTE_SECONDS) {
out.push({
id: ${p}-${f}-${Math.random().toString(36).slice(2, 8)},
pitch: p,
startTime: f * hopSec,
duration,
velocity: Math.min(1, Math.max(0.2, 0.25 + peak * 0.75)),
});
}
f = Math.max(j, f + 1);
}
}
return out;
}

/* Voice contour → note events */
/* ------------------------------------------------------------------ */

function contourToNotes(
pitches: Float32Array,
confidences: Float32Array,
rmsValues: Float32Array,
hopSec: number,
numFrames: number,
onsetThreshold: number,
): TranscribedNote[] {
// 5-point median filter to suppress vibrato flapping.
const smoothed = new Float32Array(numFrames);
const win = 5;
for (let i = 0; i < numFrames; i++) {
const vals: number[] = [];
for (let k = -2; k <= 2; k++) {
const idx = i + k;
if (idx >= 0 && idx < numFrames && confidences[idx] > 0.3 && pitches[idx] > 0) {
vals.push(pitches[idx]);
}
}
vals.sort((a, b) => a - b);
smoothed[i] = vals.length ? vals[Math.floor(vals.length / 2)] : pitches[i];
}

const voicedMin = Math.max(0.3, onsetThreshold * 0.6);
const out: TranscribedNote[] = [];
let i = 0;

while (i < numFrames) {
if (confidences[i] < voicedMin || smoothed[i] <= 0) {
i++;
continue;
}

const anchor = smoothed[i];
let j = i;
let sumPitch = 0;
let count = 0;
let sumRms = 0;
let gap = 0;

while (j < numFrames) {
const voiced = confidences[j] >= voicedMin * 0.8 && smoothed[j] > 0;
const close = voiced && Math.abs(smoothed[j] - anchor) < 1.25;

if (close) {
gap = 0;
sumPitch += smoothed[j];
sumRms += rmsValues[j];
count++;
} else {
gap++;
if (gap > 2) break;
}
j++;
}

const endFrame = j - gap;
const duration = (endFrame - i) * hopSec;

if (duration >= MIN_NOTE_SECONDS && count > 0) {
const meanPitch = sumPitch / count;
const rounded = Math.round(meanPitch);
const cents = Math.max(-100, Math.min(100, (meanPitch - rounded) * 100));
const avgRms = sumRms / count;

out.push({
id: ${rounded}-${i}-${Math.random().toString(36).slice(2, 8)},
pitch: rounded,
startTime: i * hopSec,
duration,
velocity: Math.min(1, Math.max(0.2, 0.2 + avgRms * 6)),
pitchBendCents: Math.abs(cents) > 5 ? [cents] : undefined,
});
}

i = Math.max(j, i + 1);
}

return out;
}

function mergeShortGaps(notes: TranscribedNote[]): TranscribedNote[] {
const byPitch = new Map<number, TranscribedNote[]>();
for (const n of notes) {
const arr = byPitch.get(n.pitch) ?? [];
arr.push(n);
byPitch.set(n.pitch, arr);
}

const merged: TranscribedNote[] = [];
for (const arr of byPitch.values()) {
arr.sort((a, b) => a.startTime - b.startTime);
let cur = arr[0];
for (let i = 1; i < arr.length; i++) {
const next = arr[i];
const gap = next.startTime - (cur.startTime + cur.duration);
if (gap <= SMOOTH_GAP_SECONDS) {
const total = cur.duration + Math.max(0, gap) + next.duration;
cur = {
...cur,
duration: total,
velocity: Math.max(cur.velocity, next.velocity),
};
} else {
merged.push(cur);
cur = next;
}
}
if (cur) merged.push(cur);
}

return merged.sort((a, b) => a.startTime - b.startTime || a.pitch - b.pitch);
}

function runDspEngine(
samples: Float32Array,
sampleRate: number,
settings: ConverterSettings,
): TranscribedNote[] {
const onsetThreshold = ONSET_THRESHOLD[settings.sensitivity] ?? 0.5;

if (settings.mode === 'instrument') {
const { salience, numFrames, hopSec } = buildPolySalience(samples, sampleRate);
progress(0.95);

);
}

/* ---- Voice mode: YIN on 2×-decimated signal ---- */
const decimated = new Float32Array(Math.floor(samples.length / 2));
for (let i = 0; i < decimated.length; i++) {
decimated[i] = 0.5 * (samples[i * 2] + samples[i * 2 + 1]);
}

const sr = sampleRate / 2;
const hop = 128;
const numFrames = Math.max(1, Math.floor((decimated.length - YIN_FRAME) / hop) + 1);
const hopSec = hop / sr;

const pitches = new Float32Array(numFrames);
const confs = new Float32Array(numFrames);
const rms = new Float32Array(numFrames);

for (let f = 0; f < numFrames; f++) {
const frame = decimated.subarray(f * hop, f * hop + YIN_FRAME);
const res = yinFrame(frame, sr, 65, 1000);
rms[f] = res.rms;
if (res.f0 > 0) {
pitches[f] = hzToMidi(res.f0);
confs[f] = res.confidence;
}
if (f % 200 === 0) progress(0.1 + (f / numFrames) * 0.85);
}

// Gate out silence using the RMS envelope.
let maxRms = 0;
for (let i = 0; i < numFrames; i++) maxRms = Math.max(maxRms, rms[i]);
const rmsGate = Math.max(1e-4, maxRms * 0.06);
for (let i = 0; i < numFrames; i++) {
if (rms[i] < rmsGate) confs[i] = 0;
}

progress(0.95);
return mergeShortGaps(
contourToNotes(pitches, confs, rms, hopSec, numFrames, onsetThreshold),
);
}

function decodeBasicPitch(
outputs: Record<string, { data: Float32Array; dims: readonly number[] }>,
audioDuration: number,
onsetThreshold: number,
): TranscribedNote[] {
const keys = Object.keys(outputs);
const pick = (name: string, fallbackIdx: number) => {
const hit = keys.find((k) => k.toLowerCase().includes(name));
return hit ? outputs[hit] : outputs[keys[fallbackIdx]];
};

const onset = pick('onset', 0);
const note = pick('note', 1);
if (!onset || !note) return [];

const T = onset.dims[onset.dims.length - 2];
const P = onset.dims[onset.dims.length - 1];
if (!T || !P) return [];

const fps = T / Math.max(0.001, audioDuration);
const release = onsetThreshold * 0.6;
const out: TranscribedNote[] = [];

for (let p = 0; p < P; p++) {
const midi = 21 + p;
if (midi < 0 || midi > 127) continue;

for (let t = 1; t < T - 1; t++) {
const v = onset.data[t * P + p];
if (v < onsetThreshold) continue;
// Local maximum in time → one onset per note.
if (v <= onset.data[(t - 1) * P + p] || v < onset.data[(t + 1) * P + p]) continue;

let end = t;
while (end < T - 1 && note.data[end * P + p] > release) end++;

const startTime = t / fps;
const duration = (end - t) / fps;
if (duration >= MIN_NOTE_SECONDS) {
out.push({
id: ${midi}-${t}-${Math.random().toString(36).slice(2, 8)},
pitch: midi,
startTime,
duration,
velocity: Math.min(1, Math.max(0.2, 0.25 + v * 0.75)),
});
}
t = end;
}
}

return mergeShortGaps(out);
}

IMPLEMENTATION NOTES

  1. Zero-server guarantee
    Audio never leaves the tab. decodeAudioData + OfflineAudioContext
    run in the main thread, then a copy of the PCM is transferred to the
    worker. The store keeps its own buffer for re-transcription.

  2. Main-thread fluidity
    FFT, YIN, ONNX inference, onset peak-picking, debounce and hysteresis all
    live in workers/inference.worker.ts. Only pure, cheap transforms
    (scale snap / grid quantize / key detect) run on the main thread — they are
    O(n) over a few hundred notes.

  3. Provider negotiation
    webgpu is attempted first. Any failure (missing adapter, shader
    compilation, unsupported op) is caught and the session is rebuilt with
    multithreaded wasm + SIMD. The badge in the header reflects the actual
    provider the worker reported.

  4. Graceful neural fallback
    If /models/basic-pitch.onnx 404s, or if inference yields zero notes, the
    worker runs the built-in DSP engine. The app is therefore fully functional
    out of the box with zero model assets.

  5. Sensitivity mapping (FR-2.4)
    low → 0.65 · mid → 0.50 · high → 0.35. Hysteresis uses a release
    threshold of onset × 0.6; notes under 60 ms are dropped (FR-2.5).

  6. Snap semantics (FR-3.3)
    off → raw semitones + raw contour bend preserved
    soft → pitch anchored to the scale, glide retained as pitch bend
    perfect → hard-clamped to the nearest scale degree, bends stripped

  7. Editing model
    baseNotes is the inference source of truth; notes is the corrected
    layer. Piano-roll edits write back to both so subsequent scale/grid changes
    operate on the user's intent without re-running inference.

  8. Pitch bend export
    @tonejs/midi pitch-bend support is invoked defensively inside a
    try/catch so an API drift can never block a .mid download.

  9. File System Access
    window.showSaveFilePicker() is preferred when available; a Blob URL
    download is the fallback. User cancellation (AbortError) is respected and
    does not trigger the fallback.

  10. Performance budget
    YIN runs on a 2×-decimated signal (11,025 Hz) with a fixed 512-sample
    integration window, keeping a 20 s clip well under the 5 s NFR-1 target on
    commodity hardware. The polyphonic path uses a 4096-point FFT with a
    512-sample hop.
